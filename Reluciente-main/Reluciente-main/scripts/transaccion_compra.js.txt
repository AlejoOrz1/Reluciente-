const session = db.getMongo().startSession();
session.startTransaction();

try {
    const dbSession = session.getDatabase("reluciente_db");
    
    // Preparación: Insertamos el producto inicial
    dbSession.productos.updateOne(
        { codigo: "SHAMPOO_AUTO" },
        { $setOnInsert: { nombre: "Shampoo pH Neutro", stock: 10 } },
        { upsert: true }
    );

    // Operación 1: Descontamos el stock
    const update = dbSession.productos.updateOne(
        { codigo: "SHAMPOO_AUTO", stock: { $gte: 2 } },
        { $inc: { stock: -2 } }
    );

    if (update.modifiedCount === 0) {
        throw new Error("Violación de regla de negocio: Stock insuficiente para la venta.");
    }

    // Operación 2: Generamos el ticket de venta
    dbSession.ventas.insertOne({ 
        producto: "SHAMPOO_AUTO", 
        cantidad: 2, 
        total: 5000,
        fecha: new Date()
    });

    // Confirmamos ambas operaciones
    session.commitTransaction();
    print("Transacción ACID exitosa: Stock descontado y venta registrada.");

} catch (error) {
    // Si algo falla, deshacemos todo
    session.abortTransaction();
    print("Rollback completo: La base de datos ha quedado intacta. Error: " + error.message);
} finally {
    session.endSession();
}
